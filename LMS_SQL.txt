DROP DATABASE IF EXISTS delta_LMS;

CREATE DATABASE IF NOT EXISTS delta_LMS;

USE delta_LMS;

CREATE TABLE IF NOT EXISTS users
(
	user_id int(5) NOT NULL,
	first_name varchar(50) DEFAULT NULL,
	last_name varchar(50) DEFAULT NULL,
	username varchar(20) DEFAULT NULL,
	password_hash varchar(20) DEFAULT NULL,
	role varchar(20) DEFAULT NULL,
	PRIMARY KEY (user_id)
);

CREATE TABLE IF NOT EXISTS members
(
	member_id int(5) NOT NULL,
	first_name varchar(50) DEFAULT NULL,
	last_name varchar(50) DEFAULT NULL,
	email varchar(100) DEFAULT NULL,
	phone varchar(20) DEFAULT NULL,
	PRIMARY KEY (member_id)
);

CREATE TABLE IF NOT EXISTS books
(
	book_id int(20) NOT NULL,
	ISBN varchar(50) DEFAULT NULL,
	title varchar(200) DEFAULT NULL,
	author varchar(200) DEFAULT NULL,
	genre varchar(50) DEFAULT NULL,
	call_number varchar(30) DEFAULT NULL,
	PRIMARY KEY (book_id)	
);

CREATE TABLE IF NOT EXISTS BookCopies
(
	copy_id int(20) NOT NULL,
	book_id int(20) NOT NULL,
	call_number varchar(30) DEFAULT NULL,
	status boolean DEFAULT NULL,
	PRIMARY KEY (copy_id),
	FOREIGN KEY (book_id) REFERENCES books(book_id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS transactions
(
	transaction_id int(10) NOT NULL,
	user_id int(5) DEFAULT NULL,
	member_id int(5) DEFAULT NULL,
	copy_id int(20) DEFAULT NULL,
	checkout_date date DEFAULT NULL,
	due_date date DEFAULT NULL,
	return_date date DEFAULT NULL,
	late_fee int(5) DEFAULT NULL,
	PRIMARY KEY (transaction_id),
	FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
	FOREIGN KEY (member_id) REFERENCES members(member_id) ON DELETE CASCADE,
	FOREIGN KEY (copy_id) REFERENCES BookCopies(copy_id) ON DELETE CASCADE	
);

CREATE TABLE IF NOT EXISTS holds
(
	hold_id int(10) NOT NULL,
	member_id int(5) DEFAULT NULL,
	book_id int(20) DEFAULT NULL,
	hold_date date DEFAULT NULL,
	status boolean DEFAULT NULL,
	PRIMARY KEY (hold_id),
	FOREIGN KEY (member_id) REFERENCES members(member_id) ON DELETE CASCADE,
	FOREIGN KEY (book_id) REFERENCES books(book_id) ON DELETE CASCADE	
);
